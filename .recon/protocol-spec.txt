public final class ConfigPacket extends ConfigPlayerInfoPacket {
    private UUID serverId;
    private CaptureInfo captureInfo;
    private @Nullable EncryptionInfo encryption;
    private Set<VoiceSourceLine> sourceLines;
    private Set<VoiceActivation> activations;
    private PlayerIconConfig playerIconConfig;
    public ConfigPacket(
    ) {
        super(permissions);
        this.serverId = serverId;
        this.captureInfo = captureInfo;
        this.encryption = encryption;
        this.sourceLines = sourceLines;
        this.activations = activations;
        this.playerIconConfig = playerIconConfig;
    }
    public Collection<VoiceSourceLine> getSourceLines() {
        return sourceLines;
    }
    public Collection<VoiceActivation> getActivations() {
        return activations;
    }
    public void read(ByteArrayDataInput in) throws IOException {
        this.serverId = PacketUtil.readUUID(in);
        this.captureInfo = new CaptureInfo();
        captureInfo.deserialize(in);
        if (in.readBoolean()) {
            this.encryption = new EncryptionInfo();
            encryption.deserialize(in);
        }
        this.sourceLines = Sets.newHashSet();
        int size = PacketUtil.readSafeInt(in, 0, Byte.MAX_VALUE);
        for (int i = 0; i < size; i++) {
            VoiceSourceLine sourceLine = new VoiceSourceLine();
            sourceLine.deserialize(in);
            sourceLines.add(sourceLine);
        }
        this.activations = Sets.newHashSet();
        size = PacketUtil.readSafeInt(in, 0, Byte.MAX_VALUE);
        for (int i = 0; i < size; i++) {
            VoiceActivation activation = new VoiceActivation();
            activation.deserialize(in);
            activations.add(activation);
        }
        super.read(in);
        try {
            this.playerIconConfig = new PlayerIconConfig();
            playerIconConfig.deserialize(in);
        } catch (Exception ignored) {
        }
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        PacketUtil.writeUUID(out, serverId);
        checkNotNull(captureInfo).serialize(out);
        out.writeBoolean(encryption != null);
        if (encryption != null) {
            encryption.serialize(out);
        }
        out.writeInt(sourceLines.size());
        for (VoiceSourceLine sourceLine : sourceLines) {
            sourceLine.serialize(out);
        }
        out.writeInt(activations.size());
        activations.forEach(activation -> activation.serialize(out));
        super.write(out);
        playerIconConfig.serialize(out);
    }
    public void handle(ClientPacketTcpHandler handler) {
        handler.handle(this);
    }
}

public class ConfigPlayerInfoPacket implements Packet<ClientPacketTcpHandler> {
    private Map<String, Boolean> permissions = new HashMap<>();
    public ConfigPlayerInfoPacket(@NotNull Map<String, Boolean> permissions) {
        this.permissions = checkNotNull(permissions, "playerPermissions cannot be null");
    }
    public void read(ByteArrayDataInput in) throws IOException {
        int size = PacketUtil.readSafeInt(in, 0, Byte.MAX_VALUE);
        for (int i = 0; i < size; i++) {
            String key = in.readUTF();
            boolean value = in.readBoolean();
            permissions.put(key, value);
        }
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        out.writeInt(permissions.size());
        permissions.forEach((permission, value) -> {
            out.writeUTF(permission);
            out.writeBoolean(value);
        });
    }
    public void handle(ClientPacketTcpHandler handler) {
        handler.handle(this);
    }
}

----- FILE LanguagePacket.java -----
public final class LanguagePacket implements Packet<ClientPacketTcpHandler> {
    private String languageName;
    private Map<String, String> language;
    public void read(ByteArrayDataInput in) throws IOException {
        this.languageName = in.readUTF();
        this.language = Maps.newHashMap();
        int size = PacketUtil.readSafeInt(in, 0, Short.MAX_VALUE);
        for (int i = 0; i < size; i++) {
            String key = in.readUTF();
            String value = in.readUTF();
            language.put(key, value);
        }
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        out.writeUTF(checkNotNull(languageName));
        if (language.size() > Short.MAX_VALUE) {
            throw new IllegalArgumentException("Language size is too big");
        }
        out.writeInt(language.size());
        for (Map.Entry<String, String> entry : language.entrySet()) {
            out.writeUTF(entry.getKey());
            out.writeUTF(entry.getValue());
        }
    }
    public void handle(ClientPacketTcpHandler handler) {
        handler.handle(this);
    }
}

public final class SelfAudioInfoPacket implements Packet<ClientPacketUdpHandler> {
    private UUID sourceId;
    private long sequenceNumber;
    private byte[] data;
    private short distance;
    public Optional<byte[]> getData() {
        return Optional.ofNullable(data);
    }
    public void read(ByteArrayDataInput in) throws IOException {
        this.sourceId = PacketUtil.readUUID(in);
        this.sequenceNumber = in.readLong();
        if (in.readBoolean()) {
            int length = PacketUtil.readSafeInt(in, 1, 2048);
            byte[] data = new byte[length];
            in.readFully(data);
            this.data = data;
        }
        this.distance = in.readShort();
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        PacketUtil.writeUUID(out, checkNotNull(sourceId, "sourceId"));
        out.writeLong(sequenceNumber);
        out.writeBoolean(data != null);
        if (data != null) {
            out.writeInt(data.length);
            out.write(data);
        }
        out.writeShort(distance);
    }
    public void handle(ClientPacketUdpHandler handler) {
        handler.handle(this);
    }
}

public abstract class BaseAudioPacket<T extends PacketHandler> implements Packet<T> {
    protected long sequenceNumber;
    protected byte[] data;
    public BaseAudioPacket(long sequenceNumber, byte[] data) {
        this.sequenceNumber = sequenceNumber;
        this.data = data;
        checkArgument(data.length > 0, "audio data cannot be empty");
    }
    public void read(ByteArrayDataInput in) throws IOException {
        this.sequenceNumber = in.readLong();
        int length = PacketUtil.readSafeInt(in, 1, 2048);
        byte[] data = new byte[length];
        in.readFully(data);
        this.data = data;
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        out.writeLong(sequenceNumber);
        out.writeInt(data.length);
        out.write(data);
    }
}

----- FILE PingPacket.java -----
public class PingPacket implements Packet<PacketUdpHandler> {
    private long time = System.currentTimeMillis();
    private String serverIp;
    private int serverPort;
    public PingPacket(@NotNull String serverIp, int serverPort) {
        this.serverIp = serverIp;
        this.serverPort = serverPort;
    }
    public void read(ByteArrayDataInput in) throws IOException {
        this.time = in.readLong();
        try {
            this.serverIp = PacketUtil.readSafeUTF(in, 255);
            this.serverPort = in.readUnsignedShort();
        } catch (Exception ignored) {
        }
    }
    public void write(ByteArrayDataOutput out) throws IOException {
        out.writeLong(time);
        if (serverIp != null && serverPort > 0) {
            out.writeUTF(serverIp);
            out.writeShort(serverPort);
        }
    }
    public void handle(PacketUdpHandler handler) {
        handler.handle(this);
    }
}
